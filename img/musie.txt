import { useEffect, useState } from 'react';
import {
  Heart,
  MoreHorizontal,
  Play,
  Pause,
  SkipBack,
  SkipForward,
  Repeat,
  Shuffle,
  ChevronDown,
  Moon,
  Sun,
} from 'lucide-react';
import heroThumbnail from '../assets/hero.png';
import { usePlayerStore } from '../store/usePlayerStore';
import { useVisualizer } from '../hooks/useVisualizer';
import { VolumeControl } from './VolumeControl';

interface NowPlayingProps {
  setView: (view: 'home' | 'library') => void;
}

export const NowPlaying = ({ setView }: NowPlayingProps) => {
  const [isLightMode, setIsLightMode] = useState(
    document.documentElement.classList.contains('light')
  );
  const {
    songs,
    currentTrack,
    lastTrack,
    isPlaying,
    togglePlay,
    seek,
    duration,
    setSeek,
    volume,
    setVolume,
    setTrack,
  } = usePlayerStore();

  const audioData = useVisualizer();
  const featuredTrack = currentTrack ?? lastTrack ?? songs[songs.length - 1] ?? songs ?? null;

  const formatTime = (s: number) =>
    `${Math.floor(s / 60)}:${Math.floor(s % 60)
      .toString()
      .padStart(2, '0')}`;

  const progressPercent = (seek / duration) * 100 || 0;

  useEffect(() => {
    const syncTheme = () => {
      setIsLightMode(document.documentElement.classList.contains('light'));
    };
    window.addEventListener('themechange', syncTheme);
    return () => window.removeEventListener('themechange', syncTheme);
  }, []);

  const toggleTheme = () => {
    document.documentElement.classList.toggle('light');
    window.dispatchEvent(new Event('themechange'));
  };

  // VISTA SIN TRACK SELECCIONADO (HOME / PREVIEW)
  if (!currentTrack) {
    return (
      <div className="flex-1 h-full flex flex-col items-center justify-between p-3 sm:p-4 md:p-10 pb-6 md:pb-0 relative overflow-hidden transition-colors duration-500 bg-[var(--bg-main)]">
        <div className="absolute inset-0 -z-10 opacity-20 blur-[140px] scale-150 transition-all duration-1000">
          <img
            src={featuredTrack?.coverUrl || heroThumbnail}
            alt=""
            className="w-full h-full object-cover"
          />
        </div>
        <div className="w-full max-w-4xl flex flex-col items-center justify-between gap-4 h-full">
          <div className="w-full flex justify-between items-center shrink-0 px-2 py-2">
            <button
              onClick={() => setView('library')}
              className="p-2 -ml-2 text-[var(--text-main)] opacity-40 hover:opacity-100 transition-all active:scale-90"
            >
              <ChevronDown size={32} strokeWidth={2.5} />
            </button>
            <span className="text-[10px] uppercase tracking-[0.5em] text-[var(--text-main)] opacity-30 font-black">
              Now Playing
            </span>
            <button className="p-2 -mr-2 text-[var(--text-main)] opacity-40 hover:opacity-100 transition-colors">
              <MoreHorizontal size={24} />
            </button>
          </div>

          <div className="flex-1 w-full flex flex-col items-center justify-center gap-6 min-h-0">
            <button
              onClick={() => featuredTrack && setTrack(featuredTrack)}
              className="group relative w-full max-w-[280px] sm:max-w-[320px] aspect-square rounded-[32px] sm:rounded-[40px] overflow-hidden border border-[var(--glass-border)] shadow-[0_40px_80px_rgba(0,0,0,0.35)] transition-transform duration-500 hover:scale-[1.02] active:scale-[0.98]"
            >
              <img
                src={featuredTrack?.coverUrl || heroThumbnail}
                alt={featuredTrack ? featuredTrack.title : 'Musie'}
                className="h-full w-full object-cover"
              />
              <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-black/20 to-transparent" />
              <div className="absolute left-4 right-4 bottom-4 flex items-end justify-between gap-3 text-left">
                <div className="min-w-0">
                  <p className="text-[10px] uppercase tracking-[0.5em] text-white/60 font-black">
                    {featuredTrack ? 'Featured thumbnail' : 'Home'}
                  </p>
                  <h2 className="mt-2 text-2xl font-black text-white tracking-tighter truncate">
                    {featuredTrack ? featuredTrack.title : 'Tu reproductor está listo'}
                  </h2>
                  <p className="text-sm text-white/70 font-semibold uppercase tracking-[0.25em] truncate">
                    {featuredTrack ? featuredTrack.artist : 'Entra a Library para cargar música'}
                  </p>
                </div>
                {featuredTrack ? (
                  <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-full bg-white text-black shadow-2xl transition-transform group-hover:scale-105">
                    <Play size={20} fill="currentColor" className="ml-0.5" />
                  </div>
                ) : null}
              </div>
            </button>

            <div className="max-w-md text-center space-y-3 px-6">
              <h3 className="text-[clamp(1.35rem,4vw,2rem)] font-black tracking-tighter text-[var(--text-main)]">
                Home
              </h3>
              <p className="text-sm md:text-base text-[var(--text-muted)] leading-relaxed">
                {featuredTrack
                  ? 'Toca el thumbnail para iniciar la reproducción y entrar al modo Now Playing.'
                  : 'Sube o selecciona una canción desde Library para que aparezca aquí el thumbnail inicial.'}
              </p>
              <div className="flex items-center justify-center gap-3 pt-2">
                <button
                  onClick={() => setView('library')}
                  className="px-5 py-3 rounded-full bg-[var(--text-main)] text-[var(--bg-main)] font-black text-sm uppercase tracking-[0.2em] hover:scale-105 transition-transform"
                >
                  Ir a Library
                </button>
                {featuredTrack ? (
                  <button
                    onClick={() => setTrack(featuredTrack)}
                    className="px-5 py-3 rounded-full border border-[var(--glass-border)] text-[var(--text-main)] font-black text-sm uppercase tracking-[0.2em] hover:bg-[var(--glass-bg)] transition-colors"
                  >
                    Play
                  </button>
                ) : null}
              </div>
            </div>
          </div>
        </div>
      </div>
    );
  }

  // VISTA REPRODUCIENDO (NOW PLAYING ACTIVO)
  return (
    <div className="flex-1 h-full flex flex-col items-center justify-between p-3 sm:p-4 md:p-10 pb-4 md:pb-0 relative overflow-hidden transition-colors duration-500 bg-[var(--bg-main)]">
      {/* 1. FONDO DINÁMICO */}
      <div className="absolute inset-0 -z-10 opacity-20 blur-[140px] scale-150 transition-all duration-1000">
        <img src={currentTrack.coverUrl} alt="" className="w-full h-full object-cover" />
      </div>

      {/* CONTENEDOR ADAPTATIVO: Restringe el ancho máximo en XL (`xl:max-w-2xl`) para proteger los márgenes verticales originales */}
      <div className="w-full max-w-5xl xl:max-w-4xl 2xl:max-w-4xl h-full flex flex-col landscape:flex-row xl:landscape:flex-col justify-between items-center gap-2 landscape:gap-1 xl:gap-0 pb-4 md:pb-10 min-h-0 transition-all duration-500 ease-in-out landscape:px-4 xl:landscape:px-0 mx-auto">
        {/* BLOQUE SUPERIOR (Cabecera y Portada Protagonista) */}
        <div className="w-full landscape:w-1/2 xl:landscape:w-full h-full flex flex-col justify-between items-center min-h-0 transition-all duration-500">
          {/* 2. CABECERA */}
          <div className="w-full flex justify-between items-center shrink-0 px-0 py-0">
            <button
              onClick={() => setView('library')}
              className="p-2 -ml-2 text-[var(--text-main)] opacity-40 hover:opacity-100 transition-all active:scale-90"
            >
              <ChevronDown size={32} strokeWidth={2.5} />
            </button>
            <span className="text-[10px] uppercase tracking-[0.5em] text-[var(--text-main)] opacity-30 font-black">
              Now Playing
            </span>
            <button className="p-2 -mr-2 text-[var(--text-main)] opacity-40 hover:opacity-100 transition-colors">
              <MoreHorizontal size={24} />
            </button>
          </div>

          {/* 3. PORTADA: Escalado armónico */}
          <div className="relative group flex-1 flex items-center justify-center min-h-0 w-full px-[clamp(1rem,5vw,2rem)] my-[clamp(0.25rem,2vw,1rem)]">
            <div className="absolute h-[90%] aspect-square bg-blue-600/10 rounded-full blur-[90px] opacity-80 group-hover:scale-105 transition-transform duration-1000" />
            <img
              src={currentTrack.coverUrl}
              alt={currentTrack.title}
              className="h-full max-h-[clamp(200px,50vh,620px)] landscape:max-h-[78vh] xl:max-h-[none] xl:w-full xl:max-w-[520px] 2xl:max-w-[600px] aspect-square object-cover rounded-[clamp(28px,5vw,56px)] shadow-[0_50px_100px_rgba(0,0,0,0.45)] border border-[var(--glass-border)] transition-transform duration-700"
            />
          </div>
        </div>

        {/* BLOQUE DE CONTROL INTERACTIVO: Distribución original `justify-between` recuperada con anchos estéticos */}
        <div className="w-full landscape:w-1/2 xl:landscape:w-full flex flex-col justify-between h-full xl:h-auto gap-2 landscape:gap-2 xl:gap-4 shrink-0 min-h-0 max-w-xl mx-auto">
          {/* 4. INFO PISTA CON MÁRGENES ORIGINALES */}
          <div className="w-full flex items-center justify-center px-[clamp(1rem,4vw,2rem)] landscape:px-0 xl:px-0 shrink-0 relative py-2 gap-4">
            <button
              className="flex h-10 w-10 sm:h-11 sm:w-11 items-center justify-center rounded-full bg-[var(--glass-bg)] border border-[var(--glass-border)] text-[var(--text-muted)] shadow-[var(--glass-shadow)] backdrop-blur-xl hover:text-[var(--text-main)] active:scale-90 transition-all shrink-0"
              onClick={toggleTheme}
            >
              {isLightMode ? (
                <Sun className="h-4 w-4 sm:h-5 sm:w-5 transition-colors" />
              ) : (
                <Moon className="h-4 w-4 sm:h-5 sm:w-5 transition-colors" />
              )}
            </button>

            <div className="flex flex-col items-center xl:items-center text-center flex-1 min-w-0">
              <h2 className="text-[clamp(1.1rem,4vw,1.6rem)] xl:text-2xl font-black text-[var(--text-main)] tracking-tighter leading-tight truncate w-full">
                {currentTrack.title}
              </h2>
              <p className="text-[clamp(0.6rem,1.5vw,1rem)] xl:text-sm text-blue-500 font-bold tracking-[0.2em] uppercase mt-1 opacity-80 line-clamp-1 w-full">
                {currentTrack.artist}
              </p>
            </div>

            <button className="flex h-10 w-10 sm:h-11 sm:w-11 items-center justify-center rounded-full bg-[var(--glass-bg)] border border-[var(--glass-border)] text-[var(--text-muted)] shadow-[var(--glass-shadow)] backdrop-blur-xl hover:text-red-500 active:scale-90 transition-all shrink-0">
              <Heart className="h-4 w-4 sm:h-5 sm:w-5 hover:fill-red-500 transition-colors" />
            </button>
          </div>

          {/* 5. VISUALIZER */}
          <div className="w-full px-[clamp(1rem,5vw,2rem)] landscape:px-0 shrink-0">
            <div className="relative h-[clamp(2rem,10vw,3.5rem)] landscape:h-10 xl:h-14 flex items-end justify-center gap-[clamp(2px,1vw,4px)] mb-3">
              <input
                type="range"
                min="0"
                max={duration || 0}
                step="0.1"
                value={seek}
                onChange={(e) => setSeek(parseFloat(e.target.value))}
                className="absolute bottom-0 left-0 w-full h-8 opacity-0 z-30 cursor-pointer"
              />
              {Array.from(audioData)
                .slice(0, 50)
                .map((v, i) => {
                  const isPlayed = (i / 50) * 100 < progressPercent;
                  return (
                    <div
                      key={i}
                      className={`flex-1 rounded-full transition-all duration-75 ${
                        isPlayed
                          ? 'bg-gradient-to-t from-blue-600 to-purple-500 shadow-[0_0_15px_rgba(37,99,235,0.2)]'
                          : 'bg-gray-500/20 opacity-30'
                      }`}
                      style={{ height: `${Math.max(20, (v / 255) * 100)}%` }}
                    />
                  );
                })}
            </div>

            <div className="flex justify-between items-center px-1 font-mono text-[clamp(0.6rem,1.5vw,0.8rem)] xl:text-xs text-gray-500 font-bold tracking-tighter gap-2">
              <span className="w-12 text-left shrink-0">{formatTime(seek)}</span>
              <div className="h-[1px] flex-1 mx-2 bg-gray-500/10 relative">
                <div
                  className="absolute inset-y-0 left-0 bg-blue-500/30"
                  style={{ width: `${progressPercent}%` }}
                />
              </div>
              <span className="w-12 text-right shrink-0">{formatTime(duration)}</span>
            </div>
          </div>

          {/* 6. CONTROLES DE REPRODUCCIÓN */}
          <div className="w-full max-w-sm landscape:max-w-none xl:max-w-md mx-auto flex flex-col gap-4 landscape:gap-2 xl:gap-5 px-[clamp(0.5rem,3vw,1rem)] landscape:px-0">
            <div className="flex items-center justify-between">
              <button className="p-3 text-gray-500 hover:text-blue-500 active:scale-75 transition-transform">
                <Shuffle className="h-6 w-6" />
              </button>

              <div className="flex items-center gap-8 landscape:gap-14 xl:gap-10 flex-1 justify-center">
                <button className="p-3 text-[var(--text-main)] opacity-60 hover:opacity-100 active:scale-75 transition-all">
                  <SkipBack className="h-7 w-7" fill="currentColor" />
                </button>
                <button
                  onClick={togglePlay}
                  className="flex items-center justify-center bg-[var(--text-main)] text-[var(--bg-main)] rounded-full shadow-2xl hover:scale-105 active:scale-95 transition-all h-14 w-14 landscape:h-16 landscape:w-16 xl:h-16 xl:w-16 shrink-0"
                >
                  {isPlaying ? (
                    <Pause className="h-6 w-6 landscape:h-7 landscape:w-7" fill="currentColor" />
                  ) : (
                    <Play
                      className="h-6 w-6 landscape:h-7 landscape:w-7 ml-0.5"
                      fill="currentColor"
                    />
                  )}
                </button>
                <button className="p-3 text-[var(--text-main)] opacity-60 hover:opacity-100 active:scale-75 transition-all">
                  <SkipForward className="h-7 w-7" fill="currentColor" />
                </button>
              </div>

              <button className="p-3 text-gray-500 hover:text-blue-500 active:scale-75 transition-transform">
                <Repeat className="h-6 w-6" />
              </button>
            </div>

            {/* 7. BARRA DE VOLUMEN INTEGRADA */}
            <div className="w-full pt-0">
              <VolumeControl volume={volume} onVolumeChange={setVolume} showLabel={true} />
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};
